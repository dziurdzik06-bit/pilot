            if (!id(i2c_main).is_failed()) {
              static const uint8_t addresses[] = {0x23,0x5C,0x53,0x1D,0x68,0x69,0x1E,0x0D};
              std::string detected;
              for (const uint8_t address : addresses) {
                App.feed_wdt();
                const auto result = id(i2c_main).write(address, nullptr, 0);
                ESP_LOGI("i2c_diag", "Adres 0x%02X: kod=%d", address, static_cast<int>(result));
                if (result == i2c::ERROR_OK && address != 0x23 && address != 0x5C) {
                  char text[4]; snprintf(text,sizeof(text),"%02X ",address); detected += text;
                }
              }
              if (!detected.empty()) { detected.pop_back(); id(imu_info) = "IMU I2C: " + detected; }
            } else ESP_LOGE("i2c_diag", "Blad magistrali I2C");
            {
            // The recorded chip ID is D1: native BMI160 performs its own guarded setup.
            uint8_t reg = 0x00, value = 0;
            id(imu_chip_id) = -1;
            id(imu_sample_received) = false;
            if (id(i2c_main).is_failed()) {
              id(imu_models) = "A/G: blad I2C";
              id(imu_id_line) = "I2C69: blad magistrali";
            } else {
              const auto result = id(i2c_main).write_readv(0x69, &reg, 1, &value, 1);
              if (result == i2c::ERROR_OK) {
                id(imu_chip_id) = value;
                char text[64];
                snprintf(text, sizeof(text), "69 ID00:%02X\nStan: start", value);
                id(imu_id_line) = text;
                id(imu_models) = value == 0xD1 ? "A/G: BMI160" : "A/G: Nieznany ID";
                ESP_LOGI("imu", "I2C69 register00=%02X; expected D1 for native BMI160", value);
              } else {
                id(imu_models) = "A/G: brak odpowiedzi";
                id(imu_id_line) = "I2C69: brak odpowiedzi";
                ESP_LOGW("imu", "BMI160 I2C69: probe read failed, code=%d", static_cast<int>(result));
              }
            }
            }
    - priority: -100

microphone:
  - platform: i2s_audio
    id: remote_microphone
    i2s_audio_id: microphone_i2s
    adc_type: external
    i2s_din_pin: GPIO21
    pdm: false
    channel: left
    sample_rate: 16000
    bits_per_sample: 32bit
    correct_dc_offset: false
    on_data:
      - lambda: |-
          // This callback runs on the native microphone FreeRTOS task.
          // Keep LVGL, strings, scripts and floating-point audio processing on the main loop.
          if (__atomic_load_n(&id(audio_accept_pcm), __ATOMIC_ACQUIRE) == 0U || x.size() < 4U) return;
          uint32_t peak = 0U;
          for (size_t i = 0; i + 3U < x.size(); i += 4U) {
            const uint32_t bits = static_cast<uint32_t>(x[i]) |
              (static_cast<uint32_t>(x[i + 1U]) << 8U) |
              (static_cast<uint32_t>(x[i + 2U]) << 16U) |
              (static_cast<uint32_t>(x[i + 3U]) << 24U);
            // Widen before negating: INT32_MIN has magnitude 2147483648.
            const int64_t sample = (bits & 0x80000000U) != 0U ?
              static_cast<int64_t>(bits) - 0x100000000LL : static_cast<int64_t>(bits);
            const uint64_t magnitude = sample < 0 ? static_cast<uint64_t>(-sample) : static_cast<uint64_t>(sample);
            if (magnitude > peak) peak = static_cast<uint32_t>(magnitude);
          }
          __atomic_store_n(&id(audio_peak), peak, __ATOMIC_RELAXED);
          __atomic_store_n(&id(audio_sample_ms), millis(), __ATOMIC_RELAXED);
          __atomic_fetch_add(&id(audio_sample_count), static_cast<uint32_t>(x.size() / 4U), __ATOMIC_RELEASE);
